cmake_minimum_required(VERSION 3.16)
project(MusicTitle LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
foreach(option GGML_NATIVE GGML_CPU_ALL_VARIANTS GGML_OPENMP GGML_CUDA GGML_METAL GGML_VULKAN GGML_LLAMAFILE GGML_SSE42 GGML_BMI2 GGML_AVX GGML_AVX2 GGML_FMA GGML_F16C GGML_AVX512 LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_TOOLS LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_BUILD_MTMD LLAMA_CURL LLAMA_OPENSSL)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_subdirectory(${LLAMA_SOURCE} llama)
add_executable(music-title title.cpp)
target_include_directories(music-title PRIVATE ${LLAMA_SOURCE}/vendor)
target_link_libraries(music-title PRIVATE llama)
set_target_properties(music-title PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/bin)
if(APPLE)
  enable_language(OBJCXX)
  target_sources(music-title PRIVATE ${SANDBOX_BRIDGE})
  target_link_libraries(music-title PRIVATE "-framework Foundation")
endif()
